<script>
(function () {
  /* PROBE_FRAMES: 0 = A_ascending_24, 2 = C_descending_24 */
  function probe(elId, frameIdx, forceNoInverse) {
    var f = PROBE_FRAMES[frameIdx];
    var block = SWATCH_MATRIX.decode(JSON.parse(JSON.stringify(f.block)));
    var panel = new HeatmapPanel(document.getElementById(elId));
    panel.update(block, f.spot);
    if (forceNoInverse) { panel._chart.setOption({ yAxis: { inverse: false } }); }
    var opt = panel._chart.getOption();
    var n = opt.yAxis[0].data.length;
    var px0 = panel._chart.convertToPixel({ yAxisIndex: 0 }, 0);
    var pxLast = panel._chart.convertToPixel({ yAxisIndex: 0 }, n - 1);
    var i0bottom = px0 > pxLast;
    return {
      frame: f.name,
      inverse: !!opt.yAxis[0].inverse,
      frame_first: Number(opt.yAxis[0].data[0]),
      frame_last: Number(opt.yAxis[0].data[n - 1]),
      px_index0: px0,
      px_index23: pxLast,
      screen_top: i0bottom ? Number(opt.yAxis[0].data[n - 1]) : Number(opt.yAxis[0].data[0]),
      screen_bottom: i0bottom ? Number(opt.yAxis[0].data[0]) : Number(opt.yAxis[0].data[n - 1])
    };
  }

  var out = {};
  out.before = probe("panel-a", 0, true);    /* 升序帧 + 强制无 inverse */
  out.after  = probe("panel-b", 2, false);   /* 降序帧 + 新代码默认 inverse */
  out.wrong  = probe("panel-c", 2, true);    /* 降序帧 + 无 inverse */

  out.SAME_AS_BEFORE = out.after.screen_top === out.before.screen_top
                    && out.after.screen_bottom === out.before.screen_bottom;
  out.WRONG_IS_FLIPPED = out.wrong.screen_top === out.before.screen_bottom
                      && out.wrong.screen_bottom === out.before.screen_top;

  document.getElementById("result").textContent =
    "AXIS_RESULT " + JSON.stringify(out, null, 1);
  window.PROBE_DONE = true;
})();
</script>
