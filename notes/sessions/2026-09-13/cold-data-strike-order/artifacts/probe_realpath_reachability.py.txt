"""可达性探针：冷数据键序的"非单调"在**真实生产路径**上到底能不能出现。

不是手搓 observe() —— 这里走真 StrikeWindow.rows()（含 OTM 选边、窗口滑动）
+ 真 HeatmapEngine.observe()/dump_bucket()，只把 spot 序列喂进去。

探针放工程外（工程内任何 .py 都会被 iter_py_files() 扫到 → [10] 硬编码 FAIL）。
"""
import sys
import types
from pathlib import Path

ROOT = Path("E:/US.market/SPXW SWATCH/spxw_swatch")
sys.path.insert(0, str(ROOT))

from contracts.enums import OptionRight, Quality
from contracts.tick import OptionRef
from features.heatmap_engine import HeatmapEngine
from features.strike_window import StrikeWindow

SERIAL = {"heatmap_max_buckets": 780, "heatmap_min_buckets": 2}
EACH_SIDE = 12
EXPIRY = "20260914"


def universe(lo=7500.0, hi=7900.0, step=5.0):
    """整条链的订阅集：每个行权价 P/C 两侧都在（与生产 _session_refs() 同形）。"""
    refs = []
    s = lo
    while s <= hi:
        for right in (OptionRight.PUT, OptionRight.CALL):
            refs.append(OptionRef(strike=s, right=right, expiry=EXPIRY))
        s += step
    return tuple(refs)


def cell(strike, iv):
    return types.SimpleNamespace(strike=strike, iv=iv, quality=Quality.OK)


def clock():
    c = types.SimpleNamespace()
    c.bucket_index_of_ts = lambda ts: int(ts)
    return c


def main():
    win = StrikeWindow({"heatmap_rows_each_side": EACH_SIDE})
    refs = universe()
    engine = HeatmapEngine(clock(), SERIAL)

    # spot 路径：7700 → 上涨到 7820 → 回落到 7600（跌破会话初低点）
    path = [7700.0, 7730.0, 7760.0, 7790.0, 7820.0, 7800.0, 7740.0, 7680.0,
            7640.0, 7600.0]

    print(f"订阅宇宙 {len(refs)} 个合约 · 每侧 {EACH_SIDE} 档\n")
    print(f"{'桶':>3} {'spot':>7} {'窗口首':>7} {'窗口末':>7} {'档数':>4} "
          f"{'键序单调?':>9}  键序（首3 … 末3）")

    worst = None
    for idx, spot in enumerate(path):
        rows = win.rows(refs, spot, EACH_SIDE)
        engine.observe(tuple(cell(r.strike, 0.20 + 0.01 * i)
                             for i, r in enumerate(rows)), float(idx))
        keys = list(engine.dump_bucket(idx).keys())
        mono = keys == sorted(keys)
        tail = keys[-3:] if len(keys) >= 3 else keys
        head = keys[:3]
        print(f"{idx:>3} {spot:>7.0f} {rows[0].strike:>7.0f} {rows[-1].strike:>7.0f} "
              f"{len(keys):>4} {str(mono):>9}  {head} … {tail}")
        if not mono and worst is None:
            worst = (idx, spot, keys)

    print()
    if worst is None:
        print("⇒ 该 spot 路径下键序始终单调 —— 非单调**不可达**")
        return
    idx, spot, keys = worst
    print(f"⇒ 非单调**可达**：第 {idx} 桶（spot={spot:.0f}）键序已乱")
    print(f"   完整键序: {keys}")
    print(f"   乱在尾部: {keys[-6:]}")
    print("   成因：现价曾上移到更高档（新键在末尾且更高，仍单调），"
          "随后回落到会话初低点之下 ⇒ 更低的档位被**追加到末尾**。")

    # 再写一次库，确认乱序落盘并被 recover 原样继承
    import json
    import sqlite3
    import tempfile
    from features.persistence import AsyncPersistenceWriter

    with tempfile.TemporaryDirectory() as tmp:
        db = Path(tmp) / "t.db"
        w = AsyncPersistenceWriter({"enabled": True, "db_path": str(db),
                                    "queue_maxsize": 10, "write_interval_s": 0.01})
        w._conn = sqlite3.connect(str(db))
        w._ensure_table()
        w.enqueue(idx, engine.dump_bucket(idx), False)
        w._batch_write([w._queue.get_nowait()])
        raw = w._conn.execute("select ivs_json from heatmap_buckets").fetchone()[0]
        rec = w.recover()
        w._conn.close()
        w._conn = None
    db_keys = [float(k) for k in json.loads(raw)]
    rec_keys = list(rec[0]["ivs"].keys())
    print(f"   落盘键序单调? {db_keys == sorted(db_keys)} · "
          f"recover 后单调? {rec_keys == sorted(rec_keys)} · "
          f"与内存键序一致? {rec_keys == keys}")


main()
